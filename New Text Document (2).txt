/**
 * Get the spreadsheet — works for both container-bound and standalone scripts
 */
function _getSpreadsheet() {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    if (ss) return ss;
  } catch (e) { /* ignore */ }
  return SpreadsheetApp.openById(SPREADSHEET_ID);
}

/**
 * Ensure sheet has at least N physical columns.
 * Google Sheets defaults to 26 columns — expand if HEADERS needs more.
 */
function _ensureColumns(sheet, needed) {
  var current = sheet.getMaxColumns();
  if (current < needed) {
    sheet.insertColumnsAfter(current, needed - current);
  }
}

/**
 * Initialize the sheet with headers if it doesn't exist or headers are wrong
 */
function initSheet() {
  var ss = _getSpreadsheet();
  var sheet = ss.getSheetByName(SHEET_NAME);

  if (!sheet) {
    sheet = ss.insertSheet(SHEET_NAME);
    _ensureColumns(sheet, HEADERS.length);
    sheet.getRange(1, 1, 1, HEADERS.length).setValues([HEADERS]);
    _formatHeaders(sheet);
    sheet.setFrozenRows(1);
    return sheet;
  }

  // Ensure enough physical columns exist before any range operation
  _ensureColumns(sheet, HEADERS.length);

  // Read current headers (safe — sheet definitely has >= HEADERS.length columns now)
  var existingHeaders = sheet.getRange(1, 1, 1, HEADERS.length).getValues()[0];
  var headersMatch = existingHeaders.join('|') === HEADERS.join('|');

  if (!headersMatch) {
    // Clear only the header row (up to max cols) and rewrite
    sheet.getRange(1, 1, 1, sheet.getMaxColumns()).clearContent();
    sheet.getRange(1, 1, 1, HEADERS.length).setValues([HEADERS]);
    _formatHeaders(sheet);
    sheet.setFrozenRows(1);
  }

  return sheet;
}

function _formatHeaders(sheet) {
  sheet.getRange(1, 1, 1, HEADERS.length)
    .setFontWeight('bold')
    .setBackground('#4F46E5')
    .setFontColor('#FFFFFF');
}

/**
 * Web App entry point
 */
function doGet(e) {
  initSheet();
  return HtmlService.createTemplateFromFile('Index')

    .setTitle('ระบบรับสมัครนักเรียนออนไลน์')
    .addMetaTag('viewport', 'width=device-width, initial-scale=1')
    .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
}

/**
 * Get all enrollment records — standalone, null-safe, uses openById directly
 */
function getEnrollments() {
  var out = { success: false, data: [], message: '' };
  try {
    var ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    var sheet = ss.getSheetByName(SHEET_NAME);
    if (!sheet) {
      out.success = true;
      return out;
    }

    var lastRow = sheet.getLastRow();
    if (lastRow <= 1) {
      out.success = true;
      return out;
    }

    var lastCol = sheet.getLastColumn();
    var readCols = (lastCol > 0) ? Math.min(lastCol, 38) : 1;
    var values = sheet.getRange(2, 1, lastRow - 1, readCols).getValues();

    function col(row, i) {
      if (readCols <= i || row[i] == null || row[i] === '') return '';
      var v = row[i];
      // Format Date objects properly (Google Sheets stores dates as Date)
      if (v instanceof Date) {
        return Utilities.formatDate(v, 'Asia/Bangkok', 'dd/MM/yyyy');
      }
      return String(v);
    }

    var records = [];
    for (var idx = 0; idx < values.length; idx++) {
      var row = values[idx];
      records.push({
        rowIndex: idx + 2,
        timestamp:    col(row, 0),
        prefix:       col(row, 1),
        firstName:    col(row, 2),
        lastName:     col(row, 3),
        fullName:     col(row, 1) + col(row, 2) + ' ' + col(row, 3),
        birthDate:    col(row, 4),
        religion:     col(row, 5),
        ethnicity:    col(row, 6),
        nationality:  col(row, 7),
        houseNo:      col(row, 8),
        moo:          col(row, 9),
        soi:          col(row, 10),
        road:         col(row, 11),
        subDistrict:  col(row, 12),
        district:     col(row, 13),
        province:     col(row, 14),
        zipCode:      col(row, 15),
        phone:        col(row, 16),
        email:        col(row, 17),
        fatherName:   col(row, 18),
        fatherJob:    col(row, 19),
        fatherPhone:  col(row, 20),
        motherName:   col(row, 21),
        motherJob:    col(row, 22),
        motherPhone:  col(row, 23),
        guardianType:     col(row, 24),
        guardianName:     col(row, 25),
        guardianJob:      col(row, 26),
        guardianPhone:    col(row, 27),
        guardianRelation: col(row, 28),
        prevSchool:      col(row, 29),
        prevSubDistrict: col(row, 30),
        prevDistrict:    col(row, 31),
        prevProvince:    col(row, 32),
        grade:       col(row, 33),
        studyPlan:   col(row, 34),
        photoUrl:    col(row, 35),
        documentUrl: col(row, 36),
        status:      col(row, 37) || 'รอตรวจสอบ'
      });
    }

    out.success = true;
    out.data = records;
    return out;
  } catch (error) {
    out.message = 'getEnrollments: ' + String(error);
    return out;
  }
}

/**
 * Get dashboard statistics — standalone, null-safe, no helper calls
 */
function getDashboardStats() {
  // Build result OUTSIDE try-catch so it can never be null
  var out = {
    success: false,
    total: 0,
    byGrade: {},
    byPlan: {},
    byStatus: {},
    recent: [],
    message: '',
    debug: ''
  };

  try {
    var ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    out.debug += 'ss_ok ';

    var sheet = ss.getSheetByName(SHEET_NAME);
    if (!sheet) {
      out.success = true;
      out.debug += 'no_sheet';
      return out;
    }
    out.debug += 'sheet_ok ';

    var lastRow = sheet.getLastRow();
    var lastCol = sheet.getLastColumn();
    out.debug += 'r=' + lastRow + ' c=' + lastCol + ' ';

    if (lastRow <= 1) {
      out.success = true;
      return out;
    }

    // Read only columns that physically exist (never exceed sheet bounds)
    var readCols = (lastCol > 0) ? Math.min(lastCol, 38) : 1;
    var values = sheet.getRange(2, 1, lastRow - 1, readCols).getValues();
    out.debug += 'values_ok n=' + values.length + ' ';

    var byGrade = {}, byPlan = {}, byStatus = {};
    for (var i = 0; i < values.length; i++) {
      var row = values[i];
      var grade  = (readCols > 33 && row[33] !== '' && row[33] != null) ? String(row[33]) : 'ไม่ระบุ';
      var plan   = (readCols > 34 && row[34] !== '' && row[34] != null) ? String(row[34]) : 'ไม่ระบุ';
      var status = (readCols > 37 && row[37] !== '' && row[37] != null) ? String(row[37]) : 'รอตรวจสอบ';
      byGrade[grade]   = (byGrade[grade]   || 0) + 1;
      byPlan[plan]     = (byPlan[plan]     || 0) + 1;
      byStatus[status] = (byStatus[status] || 0) + 1;
    }

    var recent = [];
    var startIdx = Math.max(0, values.length - 5);
    for (var j = values.length - 1; j >= startIdx; j--) {
      var r = values[j];
      recent.push({
        timestamp: r[0] ? String(r[0]) : '',
        fullName: String(r[1] || '') + String(r[2] || '') + ' ' + String(r[3] || ''),
        grade:     (readCols > 33 && r[33]) ? String(r[33]) : '',
        studyPlan: (readCols > 34 && r[34]) ? String(r[34]) : '',
        status:    (readCols > 37 && r[37]) ? String(r[37]) : 'รอตรวจสอบ'
      });
    }

    out.success  = true;
    out.total    = values.length;
    out.byGrade  = byGrade;
    out.byPlan   = byPlan;
    out.byStatus = byStatus;
    out.recent   = recent;
    out.debug   += 'done';

  } catch (err) {
    out.success = false;
    out.message = String(err);
    out.debug  += ' ERR:' + String(err);
  }

  return out;
}

/**
 * Upload file to Google Drive and return URL
 */
function uploadFile(fileData, fileName, mimeType) {
  try {
    const blob = Utilities.newBlob(
      Utilities.base64Decode(fileData),
      mimeType,
      fileName
    );

    const folder = getOrCreateFolder('StudentEnrollmentDocs');
    const file = folder.createFile(blob);
    file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);

    return {
      success: true,
      url: file.getUrl(),
      id: file.getId()
    };
  } catch (error) {
    return { success: false, message: error.message };
  }
}

/**
 * Get or create a folder in Drive
 */
function getOrCreateFolder(folderName) {
  const folders = DriveApp.getFoldersByName(folderName);
  if (folders.hasNext()) {
    return folders.next();
  }
  return DriveApp.createFolder(folderName);
}

/**
 * Fetch a Google Drive file as a base64 data URI (for embedding in A4 HTML).
 * Called client-side via google.script.run to avoid browser CORS restrictions.
 */
function getPhotoBase64(photoUrl) {
  try {
    if (!photoUrl) return { dataUrl: null };
    var m = photoUrl.match(/\/d\/([a-zA-Z0-9_-]+)/);
    if (!m) return { dataUrl: null };
    var file = DriveApp.getFileById(m[1]);
    var blob = file.getBlob();
    var mime = blob.getContentType() || 'image/jpeg';
    var b64  = Utilities.base64Encode(blob.getBytes());
    return { dataUrl: 'data:' + mime + ';base64,' + b64 };
  } catch (e) {
    return { dataUrl: null };
  }
}

/**
 * Batch version — accepts array of photo URLs, returns array of data URIs (same order).
 */
function getPhotosBase64(photoUrls) {
  return (photoUrls || []).map(function(url) {
    var res = getPhotoBase64(url);
    return res.dataUrl;
  });
}
