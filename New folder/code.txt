//****************************** นับจำนวนผู้สมัครแยกตามระดับชั้น ******************************//
function gradeRegistorCount(gradeRegistor) {
  let lastRow = ws1.getLastRow()
  let range = ws1.getRange("Q2:Q" + lastRow)
  let values = range.getValues()
  
  let count = 0

  for (let i = 0; i < values.length; i++) {
    if (values[i][0] === gradeRegistor) {
      count++
    }
  }
  
  return (count+1)
}

//********************* ลบไฟ์ผู้สมัครรายบุคคลด้วย recordId, colDelete(กำหนดคอลัมน์ที่ต้องการ), ชื่อ Adminที่ดำเนินการ *********************//
function deleteFile(recordId, colDelete, adminName) {
  let values = ws1.getDataRange().getDisplayValues();
  let date = new Date();

  for (var i = 0; i < values.length; i++) {
    if (values[i][0] == recordId) {
      var rowData = i + 1;
      break;
    }
  }

  let fileId = ws1.getRange(rowData, colDelete).getDisplayValue().match(/\/d\/([\w-]{33})/)[1];
  let file = DriveApp.getFileById(fileId);
  file.setTrashed(true);

  ws1.getRange(rowData, colDelete).setValue("");
  ws1.getRange(rowData, 91).setValue(adminName);
  ws1.getRange(rowData, 92).setValue(date);

  if(colDelete == 83){
    ws1.getRange(rowData, 81).setValue(ws4.getRange("B2").getDisplayValue());
  }

  return colDelete
}

//****************************** ส่งการแจ้งเตือนข้อมูลผู้สมัครใหม่ผ่าน Line Notify ******************************//
function sendLineNotify(token, formattedCount, date, stdPreName, stdFirstName, stdLastName, gradeRegistor, studyPlan){

  let formattedDate = Utilities.formatDate(date, "GMT+7", "dd/MM/yyyy")

  let  message =  "\n" +
                  "🔔 มีผู้สมัครเรียนใหม่" + "\n\n" +
                  "👉🏻 เลขประจำตัวผู้สมัคร : " + formattedCount + "\n" +
                  "📅 วันที่ : " + formattedDate + "\n" +
                  "🪪 ชื่อ - สกุล : " + stdPreName + stdFirstName + "  " + stdLastName + "\n" +
                  "💡 ชั้นเรียน : " + gradeRegistor + "\n" +
                  "🔬 แผนการเรียน : " + studyPlan
                  
  let options = {
    'method': 'post',
    "payload": {
      "message": message,
    },
    'headers': {'Authorization': 'Bearer ' + token}
  };

  UrlFetchApp.fetch('https://notify-api.line.me/api/notify', options);

}
