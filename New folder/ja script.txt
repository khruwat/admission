  function getStudyPlanDataSuccess(values){
    studyPlanData = values

    let uniqueGradeRegistor = [...new Set(studyPlanData.map(item => item[0]))]
    let studyPlanList = studyPlanData.map(item => item[2])

    let gradeRegistor = document.getElementById('gradeRegistor')
    let gradeRegistorModal = document.getElementById('gradeRegistorModal')

    let gradeRegistorFilter = document.getElementById('gradeRegistorFilter')

    for (let i = 0; i < uniqueGradeRegistor.length; i++) {
      let option = document.createElement("option");
      option.value = uniqueGradeRegistor[i];
      option.text = uniqueGradeRegistor[i];

      gradeRegistor.appendChild(option);

      let optionModal = document.createElement("option");
      optionModal.value = uniqueGradeRegistor[i];
      optionModal.text = uniqueGradeRegistor[i];

      gradeRegistorModal.appendChild(optionModal);

      let optionFilter = document.createElement("option");
      optionFilter.value = uniqueGradeRegistor[i];
      optionFilter.text = uniqueGradeRegistor[i];

      gradeRegistorFilter.appendChild(optionFilter);
    }

    let studyPlanFilter = document.getElementById('studyPlanFilter')

    for (let i = 0; i < studyPlanList.length; i++) {
      let option = document.createElement("option");
      option.value = studyPlanList[i];
      option.text = studyPlanList[i];

      studyPlanFilter.appendChild(option);

    }

  }

  //****************************** จัดรูปแบบเบอร์โทรศัพท์ 000-000-0000 ******************************//
  function formatPhoneNumber(input) {
    let phoneNumber = input.value.replace(/\D/g, '');

    if (phoneNumber.length > 3) {
      phoneNumber = phoneNumber.substring(0, 3) + '-' + phoneNumber.substring(3);
    }
    if (phoneNumber.length > 7) {
      phoneNumber = phoneNumber.substring(0, 7) + '-' + phoneNumber.substring(7);
    }

    if (phoneNumber.length > 12) {
      phoneNumber = phoneNumber.substring(0, 12);
    }

    input.value = phoneNumber;
  }

  //****************************** คำนวณอายุจากวันเกิด ******************************//
  function calculateAge(){

    let stdBirthDate = document.getElementById("stdBirthDate").value
    
    if (stdBirthDate==""){
      alert("กรุณาเลือกวันเดือนปีเกิด")
      document.getElementById("stdAge").value = "" 
    }
    else{
      let birthDateArray = stdBirthDate.split("/")

      let d1 = birthDateArray[0]
      let m1 = birthDateArray[1]
      let y1 = birthDateArray[2]

      let date = new Date()
      let d2 = date.getDate()
      let m2 = date.getMonth() + 1
      let y2 = date.getFullYear() + 543
      let month = [31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31]

      if(d1 > d2){
        d2 = d2 + month[m2 - 1]
        m2 = m2 - 1
      }

      if(m1 > m2){
        m2 = m2 + 12
        y2 = y2 - 1
      }

      let y = y2 - y1
      document.getElementById("stdAge").value = y 
    }

  }

      if(d1 > d2){
        d2 = d2 + month[m2 - 1]
        m2 = m2 - 1
      }

      if(m1 > m2){
        m2 = m2 + 12
        y2 = y2 - 1
      }

      let y = y2 - y1
      document.getElementById("stdAgeModal").value = y 
    }

  }

  //************************* โชว์ suggestion จากข้อมูลโรงเรียนเดิมที่ตั้งค่าไว้ ******************************//
  function showSuggestions(inputId, suggestions) {

    if(suggestions != null){
      const input = document.getElementById(inputId)
      const inputValue = input.value.trim().toLowerCase()
      const suggestionsContainer = document.getElementById(`suggestions-container-${inputId}`)
      
      suggestionsContainer.innerHTML = ""

      if (inputValue !== "") {
        document.getElementById('oldSchoolCheckDiv').style.display = "none"

        const matchingSuggestions = suggestions.filter(suggestion => suggestion.toLowerCase().includes(inputValue));

        matchingSuggestions.forEach(suggestion => {
          const suggestionElement = document.createElement("div")
          suggestionElement.classList.add("suggestion")
          suggestionElement.textContent = suggestion
          suggestionElement.addEventListener("click", () => {
            input.value = suggestion
            suggestionsContainer.innerHTML = ""
            suggestionsContainer.style.display = "none"
          });

          suggestionsContainer.appendChild(suggestionElement)
        });

        suggestionsContainer.style.display = "block"
      } else {
        document.getElementById('oldSchoolCheckDiv').style.display = "block"
        suggestionsContainer.style.display = "none"
      }
    }
  }

  //****************************** ตรวจสอบเลขบัตรประชาชน ******************************//
  function stdCitizenNumberCheck(stdCitizenNumber){
    if(stdCitizenNumber.length != 13){
      return true
    }

    let sum = 0
    
    for(let i = 0 ; i < 12 ; i++){
      sum += parseFloat(stdCitizenNumber.charAt(i)) * (13-i)
    }

    if(((11 - (sum % 11)) % 10 ) != parseFloat(stdCitizenNumber.charAt(12))){
      return true;
    }else{
      return false;
    }
  }

  function delFile(col){
    let recordId = document.getElementById('idModal').value
    let adminName = document.getElementById('adminNameDiv').innerHTML.split(' : ')[1].trim()
    let textDelete

    if(col == 83){
      textDelete = "ไฟล์ใบสมัคร"
    }else if(col == 85){
      textDelete = "หลักฐานการชำระเงิน"
    }else if(col == 86){
      textDelete = systemSetting[15]
    }else if(col == 87){
      textDelete = systemSetting[17]
    }else if(col == 88){
      textDelete = systemSetting[19]
    }else if(col == 89){
      textDelete = systemSetting[21]
    }else if(col == 90){
      textDelete = systemSetting[23]
    }

    Swal.fire({
      title: 'ยืนยันการลบไฟล์?',
      text: textDelete,
      icon: 'warning',
      showCancelButton: true,
      confirmButtonColor: '#198754',
      cancelButtonColor: '#d33',
      confirmButtonText: 'ลบไฟล์',
      cancelButtonText: 'ยกเลิก'
    }).then((result) => {
      if (result.isConfirmed) {
        Swal.fire({
          title: 'กำลังลบไฟล์...',
          allowOutsideClick: false,
          showConfirmButton: false
        });

        Swal.showLoading();

        google.script.run.withSuccessHandler(deleteFileSuccess).deleteFile(recordId, col, adminName)
        
      }
    })
    
  }

  function updateDataModalSubmit(obj){
    event.preventDefault()

    if(stdCitizenNumberCheck(obj.stdCitizenNumberModal.value)){
      Swal.fire({
        position: 'center',
        icon: 'error',
        title: 'โปรดตรวจสอบเลขบัตรประชาชน',
        showConfirmButton: false,
        timer: 1000
      })

      setTimeout(function(){
        document.getElementById("stdCitizenNumberModal").focus();
      }, 1500);

    }else{
      Swal.fire({
        title: 'ยืนยันการอัปเดทข้อมูล?',
        icon: 'warning',
        showCancelButton: true,
        confirmButtonColor: '#198754',
        cancelButtonColor: '#d33',
        confirmButtonText: 'บันทึก',
        cancelButtonText: 'ยกเลิก'
      }).then((result) => {
        if (result.isConfirmed) {

          if(obj.statusModal.value == "การสมัครสำเร็จ" && document.getElementById('applicationFileModal').innerHTML == '- ไม่พบไฟล์ - (เลือกสถานะการสมัครเป็น "การสมัครสำเร็จ" เพื่อสร้างไฟล์ใบสมัคร)'){

              title: 'กำลังอัปเดทข้อมูล...',
              text: 'การสร้างไฟล์ใบสมัครอาจใช้เวลา โปรดรอสักครู่...',
              allowOutsideClick: false,
              showConfirmButton: false
            });

          }else{
            Swal.fire({
              title: 'กำลังอัปเดทข้อมูล...',
              allowOutsideClick: false,
              showConfirmButton: false
            });
          }

          Swal.showLoading();

          document.getElementById("updateButton").disabled = true

          let adminName = document.getElementById('adminNameDiv').innerHTML.split(' : ')[1].trim()
          document.getElementById('updatedByModal').value = adminName

          document.getElementById('parentRelationStatusSaveModal').value = document.getElementById('parentRelationStatusModal').value
