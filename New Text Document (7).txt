// ===== Global State =====
var allEnrollments = [];
var uploadedFileData = null;
var uploadedPhotoData = null;

// ===== Navigation =====
function navigateTo(pageId) {
  document.querySelectorAll('.page-section').forEach(function(el) {
    el.classList.remove('active');
  });
  document.querySelectorAll('.nav-item').forEach(function(el) {
    el.classList.remove('active');
  });

  var target = document.getElementById('page-' + pageId);
  if (target) target.classList.add('active');

  var navItem = document.querySelector('[data-page="' + pageId + '"]');
  if (navItem) navItem.classList.add('active');

  if (pageId === 'dashboard') loadDashboard();
  else if (pageId === 'admin') loadAdminData();

  closeSidebar();
}

// ===== Mobile Sidebar =====
function toggleSidebar() {
  document.querySelector('.sidebar').classList.toggle('open');
  document.querySelector('.sidebar-overlay').classList.toggle('active');
}

function closeSidebar() {
  document.querySelector('.sidebar').classList.remove('open');
  document.querySelector('.sidebar-overlay').classList.remove('active');
}

// ===== Dashboard =====
function loadDashboard() {
  google.script.run
    .withSuccessHandler(renderDashboard)
    .withFailureHandler(handleError)
    .getDashboardStats();
}

function renderDashboard(result) {
  console.log('renderDashboard result:', JSON.stringify(result));
  if (!result || !result.success) {
    var msg = (result && result.message) ? result.message : 'ไม่สามารถโหลดข้อมูลได้';
    if (result && result.debug) msg += ' [debug: ' + result.debug + ']';
    handleError(msg);
    return;
  }

  var total    = result.total;
  var pending  = result.byStatus['รอตรวจสอบ'] || 0;
  var approved = result.byStatus['อนุมัติ'] || 0;
  var rejected = result.byStatus['ไม่อนุมัติ'] || 0;

  // Top info boxes
  animateNumber('stat-total',    total);
  animateNumber('stat-pending',  pending);
  animateNumber('stat-approved', approved);
  animateNumber('stat-rejected', rejected);

  // Bottom stats (inside chart card)
  animateNumber('bstat-total',    total);
  animateNumber('bstat-pending',  pending);
  animateNumber('bstat-approved', approved);
  animateNumber('bstat-rejected', rejected);

  // ── Grade Bar Chart (Chart.js) ──────────────────────────────────────────
  var gradeKeys = Object.keys(result.byGrade);
  var gradeVals = gradeKeys.map(function(k) { return result.byGrade[k]; });
  var COLORS = ['#6366F1','#8B5CF6','#A855F7','#EC4899','#F59E0B','#10B981','#3B82F6','#EF4444'];
  var ctx = document.getElementById('gradeChart');
  if (ctx && typeof Chart !== 'undefined') {
    if (window._gradeChartInst) { window._gradeChartInst.destroy(); }
    window._gradeChartInst = new Chart(ctx, {
      type: 'bar',
      data: {
        labels: gradeKeys.length ? gradeKeys : ['ยังไม่มีข้อมูล'],
        datasets: [{
          label: 'จำนวนผู้สมัคร',
          data: gradeVals.length ? gradeVals : [0],
          backgroundColor: COLORS.slice(0, Math.max(gradeKeys.length, 1)),
          borderRadius: 8,
          borderSkipped: false
        }]
      },
      options: {
        responsive: true,
        maintainAspectRatio: false,
        plugins: {
          legend: { display: false },
          tooltip: {
            backgroundColor: 'rgba(20,16,60,0.92)',
            titleFont: { family: 'Kanit', size: 13 },
            bodyFont:  { family: 'Kanit', size: 12 },
            callbacks: { label: function(c) { return '  ' + c.parsed.y + ' คน'; } }
          }
        },
        scales: {
          x: {
            grid: { color: 'rgba(0,0,0,0.05)' },
            ticks: { color: '#64748B', font: { family: 'Kanit', size: 11 } }
          },
          y: {
            beginAtZero: true,
            grid: { color: 'rgba(0,0,0,0.05)' },
            ticks: { color: '#64748B', font: { family: 'Kanit', size: 11 }, stepSize: 1 }
          }
        }
      }
    });
  }

  // ── Plan Distribution (Goal-Completion progress bars) ───────────────────
  var planEl   = document.getElementById('plan-distribution');
  var planKeys = Object.keys(result.byPlan || {});
  var PCOLORS  = ['#3B82F6','#EF4444','#10B981','#F59E0B','#8B5CF6','#EC4899'];
  if (planKeys.length === 0) {
    planEl.innerHTML = '<p style="color:rgba(255,255,255,0.4);text-align:center;padding:12px 0;font-size:0.85rem;">ยังไม่มีข้อมูล</p>';
  } else {
    var planMax  = Math.max.apply(null, planKeys.map(function(k) { return result.byPlan[k]; }));
    var planHtml = '<div style="display:flex;flex-direction:column;gap:14px;">';
    planKeys.forEach(function(plan, i) {
      var cnt  = result.byPlan[plan];
      var pct  = planMax > 0 ? Math.round(cnt / planMax * 100) : 0;
      var bg   = PCOLORS[i % PCOLORS.length];
      planHtml += '<div>'
        + '<div style="display:flex;justify-content:space-between;margin-bottom:5px;">'
        + '<span style="color:var(--text-primary);font-size:0.88rem;">' + escapeHtml(plan) + '</span>'
        + '<span style="color:var(--text-secondary);font-size:0.83rem;font-weight:600;">' + cnt + '/' + total + '</span>'
        + '</div>'
        + '<div style="background:#F0F2F8;border-radius:6px;height:10px;overflow:hidden;">'
        + '<div style="background:' + bg + ';height:100%;border-radius:6px;width:' + pct + '%;transition:width 0.9s ease;"></div>'
        + '</div></div>';
    });
    planHtml += '</div>';
    planEl.innerHTML = planHtml;

  // ── Grade Progress Bars ─────────────────────────────────────────────────

  if (!result.recent || result.recent.length === 0) {
    recentEl.innerHTML = '<p style="color:rgba(255,255,255,0.5);text-align:center;padding:24px;">ยังไม่มีข้อมูล</p>';
  } else {
    var AVG_COLORS = ['#7C3AED','#10B981','#F59E0B','#EF4444','#3B82F6'];
    var html = '<table class="recent-table" style="width:100%;">'
      + '<thead><tr><th></th><th>ชื่อ-นามสกุล</th><th>ระดับชั้น</th><th>แผนการเรียน</th><th>วันที่สมัคร</th><th>สถานะ</th></tr></thead><tbody>';
    result.recent.forEach(function(r, i) {
      var dateStr = r.timestamp
        ? new Date(r.timestamp).toLocaleDateString('th-TH', { day:'numeric', month:'short', year:'2-digit' })

      var letter = (r.fullName && r.fullName.trim()) ? r.fullName.trim().charAt(0) : '?';

      html += '<tr>'
        + '<td style="width:44px;"><div style="width:34px;height:34px;border-radius:50%;background:' + avBg
        + ';display:flex;align-items:center;justify-content:center;font-weight:600;font-size:0.88rem;color:#fff;">'
        + escapeHtml(letter) + '</div></td>'
        + '<td style="font-weight:500;">' + escapeHtml(r.fullName || '-') + '</td>'
        + '<td>' + escapeHtml(r.grade || '-') + '</td>'
        + '<td>' + escapeHtml(r.studyPlan || '-') + '</td>'
        + '<td style="font-size:0.8rem;color:var(--text-muted);">' + dateStr + '</td>'
        + '<td>' + getStatusBadge(r.status) + '</td>'
        + '</tr>';
    });
    html += '</tbody></table>';
    recentEl.innerHTML = html;
  }
}

function animateNumber(elementId, target) {
  var el = document.getElementById(elementId);
  if (!el) return;
  var current = 0;
  var increment = Math.ceil(target / 30) || 1;
  if (target === 0) { el.textContent = '0'; return; }
  var timer = setInterval(function() {
    current += increment;
    if (current >= target) { current = target; clearInterval(timer); }
    el.textContent = current.toLocaleString();
  }, 20);
}

// ===== Photo Upload =====
function handlePhotoSelect(inputEl) {
  var file = inputEl.files[0];
  if (!file) return;

  if (file.size > 5 * 1024 * 1024) {
    Swal.fire({ icon: 'error', title: 'ไฟล์ใหญ่เกินไป', text: 'ขนาดไฟล์ต้องไม่เกิน 5MB', confirmButtonColor: '#6366F1' });
    inputEl.value = '';
    return;
  }

  var allowedTypes = ['image/jpeg', 'image/png'];
  if (allowedTypes.indexOf(file.type) === -1) {
    Swal.fire({ icon: 'error', title: 'ประเภทไฟล์ไม่ถูกต้อง', text: 'รองรับเฉพาะ JPG, PNG', confirmButtonColor: '#6366F1' });
    inputEl.value = '';
    return;
  }

  var reader = new FileReader();
  reader.onload = function(e) {
    // Show preview in header photo box
    var previewEl = document.getElementById('photoPreview');
    if (previewEl) {
      previewEl.innerHTML = '<img src="' + e.target.result + '" style="width: 100%; height: 100%; object-fit: cover; border-radius: 8px;">';
    }
    // Show preview in upload section
    var previewLarge = document.getElementById('photoPreviewLarge');
    if (previewLarge) {
      previewLarge.innerHTML = '<img src="' + e.target.result + '" style="max-width: 100%; max-height: 150px; object-fit: contain; border-radius: 8px;">';
    }

    var base64 = e.target.result.split(',')[1];
    uploadedPhotoData = { data: base64, name: file.name, type: file.type };
  };
  reader.readAsDataURL(file);
}

// ===== File Upload =====
function handleFileSelect(inputEl) {
  var file = inputEl.files[0];
  if (!file) return;

  if (file.size > 5 * 1024 * 1024) {
    Swal.fire({ icon: 'error', title: 'ไฟล์ใหญ่เกินไป', text: 'ขนาดไฟล์ต้องไม่เกิน 5MB', confirmButtonColor: '#6366F1' });
    inputEl.value = '';
    return;
  }

  var allowedTypes = ['image/jpeg', 'image/png', 'application/pdf'];
  if (allowedTypes.indexOf(file.type) === -1) {
    Swal.fire({ icon: 'error', title: 'ประเภทไฟล์ไม่ถูกต้อง', text: 'รองรับเฉพาะ JPG, PNG, PDF', confirmButtonColor: '#6366F1' });
    inputEl.value = '';
    return;
  }

  document.getElementById('fileName').textContent = '📎 ' + file.name;
  document.getElementById('fileName').style.display = 'block';

  var reader = new FileReader();
  reader.onload = function(e) {
    var base64 = e.target.result.split(',')[1];
    uploadedFileData = { data: base64, name: file.name, type: file.type };
  };
  reader.readAsDataURL(file);
}

// Drag and drop
(function() {
  document.addEventListener('DOMContentLoaded', function() {
    var dropArea = document.getElementById('fileUploadArea');
    if (!dropArea) return;
    ['dragenter', 'dragover'].forEach(function(evt) {
      dropArea.addEventListener(evt, function(e) { e.preventDefault(); e.stopPropagation(); dropArea.classList.add('dragover'); });
    });
    ['dragleave', 'drop'].forEach(function(evt) {
      dropArea.addEventListener(evt, function(e) { e.preventDefault(); e.stopPropagation(); dropArea.classList.remove('dragover'); });
    });
    dropArea.addEventListener('drop', function(e) {
      var files = e.dataTransfer.files;
      if (files.length > 0) {
        document.getElementById('fileInput').files = files;
        handleFileSelect(document.getElementById('fileInput'));
      }
    });
  });
})();

// ===== Enrollment Form =====
function submitEnrollForm() {
  if (!validateForm()) return;

  Swal.fire({
    title: 'กำลังบันทึกข้อมูล...',
    html: 'กรุณารอสักครู่',
    allowOutsideClick: false,
    allowEscapeKey: false,
    didOpen: function() { Swal.showLoading(); }
  });

  // Upload photo first, then document, then submit
  uploadPhotoThen(function(photoUrl) {
    uploadDocThen(function(docUrl) {
      submitFormData(photoUrl, docUrl);
    });
  });
}

function uploadPhotoThen(callback) {
  if (uploadedPhotoData) {
    google.script.run
      .withSuccessHandler(function(res) {
        callback(res.success ? res.url : '');
      })
      .withFailureHandler(function() { callback(''); })
      .uploadFile(uploadedPhotoData.data, uploadedPhotoData.name, uploadedPhotoData.type);
  } else {
    callback('');
  }
}

function uploadDocThen(callback) {
  if (uploadedFileData) {
    google.script.run
      .withSuccessHandler(function(res) {
        callback(res.success ? res.url : '');
      })
      .withFailureHandler(function() { callback(''); })
      .uploadFile(uploadedFileData.data, uploadedFileData.name, uploadedFileData.type);
  } else {
    callback('');
  }
}

  var guardianType = 'บิดา';

  }

  var planRadios = document.getElementsByName('studyPlan');
  for (var j = 0; j < planRadios.length; j++) {
    if (planRadios[j].checked) { studyPlan = planRadios[j].value; break; }
  }

  var data = {
    prefix: val('prefix'),
    firstName: val('firstName'),
    lastName: val('lastName'),
    birthDate: val('birthDate'),
    religion: val('religion'),
    ethnicity: val('ethnicity'),
    nationality: val('nationality'),
    houseNo: val('houseNo'),
    moo: val('moo'),
    soi: val('soi'),
    road: val('road'),
    subDistrict: val('subDistrict'),
    district: val('district'),
    province: val('province'),
    zipCode: val('zipCode'),
    phone: val('phone'),
    email: val('email'),
    fatherName: val('fatherName'),
    fatherJob: val('fatherJob'),
    fatherPhone: val('fatherPhone'),
    motherName: val('motherName'),
    motherJob: val('motherJob'),
    motherPhone: val('motherPhone'),
    guardianType: guardianType,
    guardianName: val('guardianName'),
    guardianJob: val('guardianJob'),
    guardianPhone: val('guardianPhone'),
    guardianRelation: val('guardianRelation'),
    prevSchool: val('prevSchool'),
    prevSubDistrict: val('prevSubDistrict'),
    prevDistrict: val('prevDistrict'),
    prevProvince: val('prevProvince'),
    grade: val('grade'),
    studyPlan: studyPlan,
    photoUrl: photoUrl,
    documentUrl: documentUrl
  };

  google.script.run
    .withSuccessHandler(function(result) {
      if (result.success) {
        Swal.fire({ icon: 'success', title: 'สำเร็จ!', text: result.message, confirmButtonColor: '#6366F1' });
        resetForm();
      } else {
        Swal.fire({ icon: 'error', title: 'เกิดข้อผิดพลาด', text: result.message, confirmButtonColor: '#6366F1' });
      }
    })
    .withFailureHandler(handleError)
    .processForm(data);
}

function val(id) {
  var el = document.getElementById(id);
  return el ? el.value.trim() : '';
}

function validateForm() {
  var isValid = true;
  var requiredFields = ['prefix', 'firstName', 'lastName', 'grade', 'phone'];

  document.querySelectorAll('.form-input.error').forEach(function(el) {
    el.classList.remove('error');
  });

  requiredFields.forEach(function(id) {
    var el = document.getElementById(id);
    if (!el || !el.value.trim()) {
      if (el) el.classList.add('error');
      isValid = false;
    }
  });

  // Phone validation
  var phoneVal = val('phone');
  if (phoneVal && !/^[0-9]{9,10}$/.test(phoneVal)) {
    document.getElementById('phone').classList.add('error');
    isValid = false;
  }

  // Email validation (optional)
  var emailVal = val('email');
  if (emailVal && !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(emailVal)) {
    document.getElementById('email').classList.add('error');
    isValid = false;
  }

  if (!isValid) {
    Swal.fire({ icon: 'warning', title: 'กรุณากรอกข้อมูลให้ครบถ้วน', text: 'ตรวจสอบข้อมูลที่มีเครื่องหมาย * อีกครั้ง', confirmButtonColor: '#6366F1' });
  }

  return isValid;
}

function resetForm() {
  document.getElementById('enrollForm').reset();
  document.querySelectorAll('.form-input.error').forEach(function(el) { el.classList.remove('error'); });
  document.getElementById('fileName').style.display = 'none';
  var photoPreview = document.getElementById('photoPreview');
  if (photoPreview) {
    photoPreview.innerHTML = '<i class="fas fa-camera" style="font-size: 1.5rem; color: #94A3B8;"></i><span style="font-size: 0.7rem; color: #94A3B8; margin-top: 4px;">ติดรูปถ่าย<br>1 นิ้ว</span>';
  }
  var photoPreviewLarge = document.getElementById('photoPreviewLarge');
  if (photoPreviewLarge) {
    photoPreviewLarge.innerHTML = '<i class="fas fa-user-circle" style="font-size: 3rem; color: #CBD5E1;"></i><p style="font-size: 0.75rem; color: #94A3B8; margin-top: 6px;">คลิกเพื่ออัปโหลดรูปถ่าย</p>';
  }
  uploadedFileData = null;
  uploadedPhotoData = null;
}

// ===== Admin / Data Table =====
function loadAdminData() {
  google.script.run
    .withSuccessHandler(renderAdminTable)
    .withFailureHandler(handleError)
    .getEnrollments();
}

function renderAdminTable(result) {
  if (!result || !result.success) {
    var msg = (result && result.message) ? result.message : 'ไม่สามารถโหลดข้อมูลได้';
    handleError(msg);
    return;
  }
  allEnrollments = result.data;
  renderFilteredTable(allEnrollments);
}

function renderFilteredTable(data) {
  var tbody = document.getElementById('tableBody');

  if (data.length === 0) {
    tbody.innerHTML = '<tr><td colspan="8" style="text-align: center; padding: 40px; color: var(--text-secondary);"><i class="fas fa-inbox" style="font-size: 2rem; margin-bottom: 10px; display: block; color: #CBD5E1;"></i>ไม่พบข้อมูล</td></tr>';
    document.getElementById('recordCount').textContent = 'แสดง 0 รายการ';
    return;
  }

  var html = '';
  data.forEach(function(row, idx) {
    var displayName = escapeHtml(row.prefix) + escapeHtml(row.firstName) + ' ' + escapeHtml(row.lastName);
    html += '<tr>'
      + '<td>' + (idx + 1) + '</td>'
      + '<td>' + escapeHtml(String(row.timestamp)) + '</td>'
      + '<td>' + displayName + '</td>'
      + '<td>' + escapeHtml(row.grade) + '</td>'
      + '<td>' + escapeHtml(row.studyPlan) + '</td>'
      + '<td>' + escapeHtml(row.phone) + '</td>'
      + '<td>' + getStatusBadge(row.status) + '</td>'
      + '<td>'
      + '<div style="display: flex; gap: 4px; flex-wrap: wrap;">'
      + '<button class="btn btn-warning btn-sm" onclick="editEnrollment(' + idx + ')" title="แก้ไขข้อมูล" style="background:#F59E0B;color:#fff;"><i class="fas fa-pencil-alt"></i></button>'
      + '<button class="btn btn-primary btn-sm" onclick="viewA4(' + idx + ')" title="ดูแบบ A4"><i class="fas fa-eye"></i></button>'
      + '<button class="btn btn-success btn-sm" onclick="changeStatus(' + row.rowIndex + ', \'อนุมัติ\')" title="อนุมัติ"><i class="fas fa-check"></i></button>'
      + '<button class="btn btn-danger btn-sm" onclick="changeStatus(' + row.rowIndex + ', \'ไม่อนุมัติ\')" title="ไม่อนุมัติ"><i class="fas fa-times"></i></button>'
      + '<button class="btn btn-sm" style="background:#64748B;color:#FFF;" onclick="confirmDelete(' + row.rowIndex + ')" title="ลบ"><i class="fas fa-trash"></i></button>'
      + '</div></td></tr>';
  });

  tbody.innerHTML = html;
  document.getElementById('recordCount').textContent = 'แสดง ' + data.length + ' รายการ';
}

function filterTable() {
  var search = document.getElementById('searchInput').value.toLowerCase();
  var gradeFilter = document.getElementById('filterGrade').value;
  var statusFilter = document.getElementById('filterStatus').value;

  var filtered = allEnrollments.filter(function(row) {
    var nameStr = (row.firstName || '') + ' ' + (row.lastName || '');
    var matchSearch = !search || nameStr.toLowerCase().indexOf(search) !== -1 || (row.phone && row.phone.indexOf(search) !== -1);
    var matchGrade = !gradeFilter || row.grade === gradeFilter;
    var matchStatus = !statusFilter || row.status === statusFilter;
    return matchSearch && matchGrade && matchStatus;
  });

  renderFilteredTable(filtered);
}

function changeStatus(rowIndex, newStatus) {
  Swal.fire({
    title: 'ยืนยันการเปลี่ยนสถานะ?',
    text: 'เปลี่ยนเป็น "' + newStatus + '"',
    icon: 'question',
    showCancelButton: true,
    confirmButtonColor: '#6366F1',
    cancelButtonColor: '#94A3B8',
    confirmButtonText: 'ยืนยัน',
    cancelButtonText: 'ยกเลิก'
  }).then(function(result) {
    if (result.isConfirmed) {
      Swal.fire({ title: 'กำลังอัปเดต...', allowOutsideClick: false, didOpen: function() { Swal.showLoading(); } });
      google.script.run
        .withSuccessHandler(function(res) {
          if (res.success) { Swal.fire({ icon: 'success', title: 'สำเร็จ!', text: res.message, confirmButtonColor: '#6366F1' }); loadAdminData(); }
          else { Swal.fire({ icon: 'error', title: 'ผิดพลาด', text: res.message, confirmButtonColor: '#6366F1' }); }
        })
        .withFailureHandler(handleError)
        .updateStatus(rowIndex, newStatus);
    }
  });
}

function confirmDelete(rowIndex) {
  Swal.fire({
    title: 'ยืนยันการลบ?',

function _a4Esc(text) {
  if (!text) return '';
  return String(text)
    .replace(/&/g,'&amp;').replace(/</g,'&lt;')

}

function viewA4(idx) {
  var d = allEnrollments[idx];
  if (!d) return;

  function openPreview(enriched) {
    Swal.fire({
      title: 'ดูใบสมัคร — ' + escapeHtml(d.firstName+' '+d.lastName),
      html: '<iframe id="a4frame" style="width:100%;height:500px;border:none;background:#fff;"></iframe>',
      width: '860px',
      showCancelButton: true,
      confirmButtonText: '<i class="fas fa-print"></i> พิมพ์',
      cancelButtonText: 'ปิด',
      confirmButtonColor: '#7C3AED',
      didOpen: function() {
        var frame = document.getElementById('a4frame');
        var fdoc = frame.contentDocument || frame.contentWindow.document;
        fdoc.open();
        fdoc.write('<!DOCTYPE html><html><head><meta charset="utf-8"><style>'+_a4Css()+'</style></head><body>'+_a4Html(enriched)+'</body></html>');
        fdoc.close();
      }
    }).then(function(r) {
      if (r.isConfirmed) printSingleA4(enriched);
    });
  }

  if (d.photoUrl) {
    google.script.run
      .withSuccessHandler(function(res) {
        openPreview(Object.assign({}, d, {photoBase64: res && res.dataUrl ? res.dataUrl : null}));
      })
      .withFailureHandler(function() { openPreview(d); })
      .getPhotoBase64(d.photoUrl);
  } else {
    openPreview(d);
  }
}

function printSingleA4(d) {
  var win = window.open('', '_blank', 'width=900,height=700,scrollbars=yes');
  win.document.write([
    '<!DOCTYPE html><html><head>',
    '<meta charset="utf-8">',
    '<title>ใบสมัคร — '+_a4Esc(d.firstName)+' '+_a4Esc(d.lastName)+'</title>',
    '<style>'+_a4Css()+'</style>',
    '</head><body>',
    _a4Html(d),
    '<scr'+'ipt>window.onload=function(){window.print();};<\/scr'+'ipt>',
    '</body></html>'
  ].join(''));
  win.document.close();
}

function printA4() {
  if (!allEnrollments || allEnrollments.length === 0) {
    Swal.fire({ icon: 'info', title: 'ไม่มีข้อมูล', text: 'ยังไม่มีข้อมูลผู้สมัคร', confirmButtonColor: '#7C3AED' });
    return;
  }
  function doWrite(records) {
    var win = window.open('', '_blank', 'width=900,height=700,scrollbars=yes');
    var pages = records.map(function(d) {
      return '<div style="page-break-after:always;">'+_a4Html(d)+'</div>';
    }).join('');
    win.document.write([
      '<!DOCTYPE html><html><head>',
      '<meta charset="utf-8"><title>ใบสมัครทั้งหมด</title>',
      '<style>'+_a4Css()+'</style>',
      '</head><body>',
      pages,
      '<scr'+'ipt>window.onload=function(){window.print();};<\/scr'+'ipt>',
      '</body></html>'
    ].join(''));
    win.document.close();
  }
  var photoUrls = allEnrollments.map(function(d) { return d.photoUrl || ''; });
  google.script.run
    .withSuccessHandler(function(base64List) {
      var enriched = allEnrollments.map(function(d, i) {
        return Object.assign({}, d, {photoBase64: base64List[i] || null});
      });
      doWrite(enriched);
    })
    .withFailureHandler(function() { doWrite(allEnrollments); })
    .getPhotosBase64(photoUrls);
}

function exportToExcel() {
  if (allEnrollments.length === 0) {
    Swal.fire({ icon: 'info', title: 'ไม่มีข้อมูล', confirmButtonColor: '#6366F1' });
    return;
  }

  var headers = ['#','วันที่','คำนำหน้า','ชื่อ','นามสกุล','วันเกิด','ศาสนา','เชื้อชาติ','สัญชาติ','เบอร์โทร','อีเมล','ระดับชั้น','แผนการเรียน','โรงเรียนเดิม','ชื่อบิดา','ชื่อมารดา','ผู้ปกครอง','สถานะ'];
  var csv = [headers.map(function(h) { return '"' + h + '"'; }).join(',')];

  allEnrollments.forEach(function(row, idx) {
    var line = [
      idx + 1,
      '"' + String(row.timestamp).replace(/"/g, '""') + '"',
      '"' + (row.prefix || '').replace(/"/g, '""') + '"',
      '"' + (row.firstName || '').replace(/"/g, '""') + '"',
      '"' + (row.lastName || '').replace(/"/g, '""') + '"',
      '"' + (row.birthDate || '').replace(/"/g, '""') + '"',
      '"' + (row.religion || '').replace(/"/g, '""') + '"',
      '"' + (row.ethnicity || '').replace(/"/g, '""') + '"',
      '"' + (row.nationality || '').replace(/"/g, '""') + '"',
      '"' + (row.phone || '').replace(/"/g, '""') + '"',
      '"' + (row.email || '').replace(/"/g, '""') + '"',
      '"' + (row.grade || '').replace(/"/g, '""') + '"',
      '"' + (row.studyPlan || '').replace(/"/g, '""') + '"',
      '"' + (row.prevSchool || '').replace(/"/g, '""') + '"',
      '"' + (row.fatherName || '').replace(/"/g, '""') + '"',
      '"' + (row.motherName || '').replace(/"/g, '""') + '"',
      '"' + (row.guardianName || '').replace(/"/g, '""') + '"',
      '"' + (row.status || '').replace(/"/g, '""') + '"'
    ];
    csv.push(line.join(','));
  });

  var csvContent = '\uFEFF' + csv.join('\n');
  var blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' });
  var url = URL.createObjectURL(blob);
  var link = document.createElement('a');
  link.href = url;
  link.download = 'enrollment_data_' + formatDate(new Date()) + '.csv';
  link.click();
  URL.revokeObjectURL(url);

  Swal.fire({ icon: 'success', title: 'ส่งออกสำเร็จ!', text: 'ดาวน์โหลดไฟล์ CSV แล้ว', confirmButtonColor: '#6366F1', timer: 3000 });
}

// ===== Utility =====
function escapeHtml(text) {
  if (!text) return '';
  var div = document.createElement('div');
  div.appendChild(document.createTextNode(String(text)));
  return div.innerHTML;
}

function getStatusBadge(status) {
  var cls = 'badge-pending';
  if (status === 'อนุมัติ') cls = 'badge-approved';
  else if (status === 'ไม่อนุมัติ') cls = 'badge-rejected';
  return '<span class="badge ' + cls + '">' + escapeHtml(status || 'รอตรวจสอบ') + '</span>';
}

function formatDate(date) {
  var d = date.getDate().toString().padStart(2, '0');
  var m = (date.getMonth() + 1).toString().padStart(2, '0');
  return date.getFullYear() + '-' + m + '-' + d;
}

}
